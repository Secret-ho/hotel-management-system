-- =====================================================
-- مرحله 1: ساخت دیتابیس
-- =====================================================
CREATE DATABASE IF NOT EXISTS HotelManagementSystem;
USE HotelManagementSystem;
-- =====================================================

-- 1. جدول هتل (Hotel) - از PDF اول
CREATE TABLE Hotel (
    hotel_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    code VARCHAR(20) UNIQUE NOT NULL,
    address TEXT,
    phone_number VARCHAR(20),
    email VARCHAR(100),
    total_floors INT,
    total_rooms INT,
    star_rating INT CHECK (star_rating BETWEEN 1 AND 5),
    facilities TEXT,
    manager_name VARCHAR(100),
    website VARCHAR(150)
);

-- 2. جدول امکانات هتل (Hotel_facility) - ویژگی چندمقداری از فایل دوم
CREATE TABLE Hotel_facility (
    hotel_id INT NOT NULL,
    facility_name VARCHAR(100) NOT NULL,
    PRIMARY KEY (hotel_id, facility_name),
    FOREIGN KEY (hotel_id) REFERENCES Hotel(hotel_id) ON DELETE CASCADE
);

-- 3. جدول تلفن (Phone) - ویژگی چندمقداری برای Hotel, Customer, Staff از فایل دوم
CREATE TABLE Phone (
    phone_id INT PRIMARY KEY AUTO_INCREMENT,
    phone_number VARCHAR(20) NOT NULL,
    person_id INT NOT NULL,
    person_type ENUM('hotel', 'customer', 'staff') NOT NULL
);

-- 4. جدول نوع اتاق (RoomType) - از PDF اول
CREATE TABLE RoomType (
    room_type_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    description TEXT,
    default_capacity INT,
    min_price_per_night DECIMAL(10,2),
    max_price_per_night DECIMAL(10,2),
    min_booking_count INT,
    max_booking_count INT,
    status ENUM('available', 'unavailable') DEFAULT 'available'
);

-- 5. جدول اتاق (Room)
CREATE TABLE Room (
    room_id INT PRIMARY KEY AUTO_INCREMENT,
    room_number VARCHAR(10) NOT NULL,
    floor_number INT,
    status ENUM('empty', 'reserved', 'occupied', 'cleaning', 'out_of_service') DEFAULT 'empty',
    standard_capacity INT,
    max_capacity INT,
    standard_price_per_night DECIMAL(10,2),
    amenities TEXT,
    room_type_id INT,
    hotel_id INT,
    FOREIGN KEY (room_type_id) REFERENCES RoomType(room_type_id) ON DELETE SET NULL,
    FOREIGN KEY (hotel_id) REFERENCES Hotel(hotel_id) ON DELETE CASCADE
);

-- 6. جدول مشتری (Customer) 
CREATE TABLE Customer (
    customer_id INT PRIMARY KEY AUTO_INCREMENT,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    national_id VARCHAR(20) UNIQUE NOT NULL,
    birth_date DATE,
    gender ENUM('male', 'female', 'other'),
    mobile_phone VARCHAR(20),
    landline_phone VARCHAR(20),
    email VARCHAR(100),
    address TEXT,
    first_reservation_date DATE,
    total_stays INT DEFAULT 0,
    membership_level ENUM('standard', 'silver', 'gold') DEFAULT 'standard',
    registration_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    profile_picture VARCHAR(255)
);

-- 7. جدول کارکنان (Staff) 
CREATE TABLE Staff (
    staff_id INT PRIMARY KEY AUTO_INCREMENT,
    personal_code VARCHAR(20) UNIQUE NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    national_id VARCHAR(20) UNIQUE NOT NULL,
    birth_date DATE,
    address TEXT,
    job_title VARCHAR(50),
    access_level ENUM('admin', 'reception', 'housekeeping', 'cashier') NOT NULL,
    hire_date DATE,
    base_salary DECIMAL(10,2),
    bank_account_info TEXT,
    employment_status ENUM('active', 'on_leave', 'terminated') DEFAULT 'active',
    hotel_id INT,
    FOREIGN KEY (hotel_id) REFERENCES Hotel(hotel_id) ON DELETE SET NULL
);

-- 8. جدول مدیر (Manager) 
CREATE TABLE Manager (
    staff_id INT PRIMARY KEY,
    management_level VARCHAR(50),
    bonus DECIMAL(10,2),
    FOREIGN KEY (staff_id) REFERENCES Staff(staff_id) ON DELETE CASCADE
);

-- 9. جدول پذیرش (Receptionist) 
CREATE TABLE Receptionist (
    staff_id INT PRIMARY KEY,
    shift ENUM('morning', 'evening', 'night'),
    desk_number INT,
    FOREIGN KEY (staff_id) REFERENCES Staff(staff_id) ON DELETE CASCADE
);

-- 10. جدول نظافت (Housekeeping) 
CREATE TABLE Housekeeping (
    staff_id INT PRIMARY KEY,
    area_responsibility VARCHAR(100),
    tools_assigned TEXT,
    FOREIGN KEY (staff_id) REFERENCES Staff(staff_id) ON DELETE CASCADE
);

-- 11. جدول رزرو (Reservation)
CREATE TABLE Reservation (
    reservation_id INT PRIMARY KEY AUTO_INCREMENT,
    reservation_number VARCHAR(20) UNIQUE NOT NULL,
    reservation_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    check_in_date DATE NOT NULL,
    check_out_date DATE NOT NULL,
    nights INT,
    total_guests INT,
    adults INT,
    children INT,
    status ENUM('confirmed', 'pending', 'cancelled', 'completed') DEFAULT 'pending',
    estimated_total DECIMAL(10,2),
    booking_method ENUM('online', 'phone', 'in_person') DEFAULT 'online',
    discount_code VARCHAR(50),
    room_id INT,
    customer_id INT,
    FOREIGN KEY (room_id) REFERENCES Room(room_id) ON DELETE SET NULL,
    FOREIGN KEY (customer_id) REFERENCES Customer(customer_id) ON DELETE CASCADE
);

-- 12. جدول پرداخت (Payment) 
CREATE TABLE Payment (
    payment_id INT PRIMARY KEY AUTO_INCREMENT,
    payment_number VARCHAR(20) UNIQUE NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    payment_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    payment_method ENUM('online', 'card', 'cash', 'bank_transfer') NOT NULL,
    transaction_tracking_id VARCHAR(100),
    status ENUM('successful', 'failed', 'pending') DEFAULT 'pending',
    source_account_name VARCHAR(100),
    reservation_id INT,
    FOREIGN KEY (reservation_id) REFERENCES Reservation(reservation_id) ON DELETE SET NULL
);

-- 13. جدول فاکتور (Invoice) 
CREATE TABLE Invoice (
    invoice_id INT PRIMARY KEY AUTO_INCREMENT,
    invoice_number VARCHAR(20) UNIQUE NOT NULL,
    issue_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    grand_total DECIMAL(10,2),
    paid_amount DECIMAL(10,2),
    remaining_amount DECIMAL(10,2),
    cost_details TEXT,
    status ENUM('settled', 'unsettled') DEFAULT 'unsettled',
    reservation_id INT,
    customer_id INT,
    issued_by_staff_id INT,
    FOREIGN KEY (reservation_id) REFERENCES Reservation(reservation_id) ON DELETE SET NULL,
    FOREIGN KEY (customer_id) REFERENCES Customer(customer_id) ON DELETE CASCADE,
    FOREIGN KEY (issued_by_staff_id) REFERENCES Staff(staff_id) ON DELETE SET NULL
);

-- 14. جدول فاکتور شرکتی (Corporate_Invoice) 
CREATE TABLE Corporate_Invoice (
    invoice_id INT PRIMARY KEY,
    company_name VARCHAR(100) NOT NULL,
    company_registration_number VARCHAR(50),
    FOREIGN KEY (invoice_id) REFERENCES Invoice(invoice_id) ON DELETE CASCADE
);

-- 15. جدول فاکتور شخصی (Individual_Invoice) 
CREATE TABLE Individual_Invoice (
    invoice_id INT PRIMARY KEY,
    national_id_customer VARCHAR(20),
    FOREIGN KEY (invoice_id) REFERENCES Invoice(invoice_id) ON DELETE CASCADE
);

-- 16. جدول خدمات (Service) 
CREATE TABLE Service (
    service_id INT PRIMARY KEY AUTO_INCREMENT,
    service_name VARCHAR(100) NOT NULL,
    description TEXT,
    base_fee DECIMAL(10,2),
    unit_of_measure ENUM('hourly', 'count', 'kg') DEFAULT 'count',
    service_hours VARCHAR(100),
    responsible_staff_id INT,
    room_id INT,
    customer_id INT,
    reservation_id INT,
    FOREIGN KEY (responsible_staff_id) REFERENCES Staff(staff_id) ON DELETE SET NULL,
    FOREIGN KEY (room_id) REFERENCES Room(room_id) ON DELETE SET NULL,
    FOREIGN KEY (customer_id) REFERENCES Customer(customer_id) ON DELETE CASCADE,
    FOREIGN KEY (reservation_id) REFERENCES Reservation(reservation_id) ON DELETE CASCADE
);

-- 17. جدول بازخورد (Feedback)
CREATE TABLE Feedback (
    feedback_id INT PRIMARY KEY AUTO_INCREMENT,
    customer_id INT NOT NULL,
    reservation_id INT,
    feedback_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    feedback_text TEXT,
    overall_rating INT CHECK (overall_rating BETWEEN 1 AND 5),
    room_quality_rating INT CHECK (room_quality_rating BETWEEN 1 AND 5),
    service_quality_rating INT CHECK (service_quality_rating BETWEEN 1 AND 5),
    staff_behavior_rating INT CHECK (staff_behavior_rating BETWEEN 1 AND 5),
    status ENUM('pending', 'processing', 'done') DEFAULT 'pending',
    feedback_status ENUM('pending', 'approved', 'rejected') DEFAULT 'pending',
    submission_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    feedback_content TEXT,
    FOREIGN KEY (customer_id) REFERENCES Customer(customer_id) ON DELETE CASCADE,
    FOREIGN KEY (reservation_id) REFERENCES Reservation(reservation_id) ON DELETE SET NULL
);
